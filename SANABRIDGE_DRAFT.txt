SanaBridge — HackAlem AI MVP Draft
==================================

Назначение
----------
SanaBridge превращает короткое и неполное описание бизнес-проблемы в структурированный AI Challenge:
1) OpenAI извлекает известные факты и определяет пробелы.
2) Система задаёт контекстные уточняющие вопросы.
3) Backend детерминированно считает Readiness Score.
4) Бизнес дополняет задачу и публикует её.
5) Студенческие команды видят каталог и отправляют заявки.
6) Бизнес выбирает команду.
7) NVIDIA NIM опционально выполняет независимый технический review.

ВАЖНО: этот TXT — черновой blueprint для Codex, а не готовое production-приложение.
Никогда не публикуйте реальные API-ключи в GitHub.

        db.close()

from datetime import datetime
from sqlalchemy import DateTime, ForeignKey, Integer, String, Text
from sqlalchemy.orm import Mapped, mapped_column, relationship
from .database import Base

class Challenge(Base):
    __tablename__ = "challenges"

    id: Mapped[int] = mapped_column(Integer, primary_key=True)
    title: Mapped[str | None] = mapped_column(String(255), nullable=True)
    raw_description: Mapped[str] = mapped_column(Text)

    problem: Mapped[str | None] = mapped_column(Text, nullable=True)
    goal: Mapped[str | None] = mapped_column(Text, nullable=True)
    target_users: Mapped[str | None] = mapped_column(Text, nullable=True)
    available_data: Mapped[str | None] = mapped_column(Text, nullable=True)
    expected_result: Mapped[str | None] = mapped_column(Text, nullable=True)
    success_metrics: Mapped[str | None] = mapped_column(Text, nullable=True)
    constraints: Mapped[str | None] = mapped_column(Text, nullable=True)
    deadline: Mapped[str | None] = mapped_column(String(100), nullable=True)

    required_skills: Mapped[str | None] = mapped_column(Text, nullable=True)
    suggested_solution: Mapped[str | None] = mapped_column(Text, nullable=True)

    readiness_score: Mapped[int] = mapped_column(Integer, default=0)
    status: Mapped[str] = mapped_column(String(30), default="DRAFT")
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.utcnow)

    applications: Mapped[list["Application"]] = relationship(
        back_populates="challenge",
        cascade="all, delete-orphan",
    )

class Application(Base):
    __tablename__ = "applications"

    id: Mapped[int] = mapped_column(Integer, primary_key=True)
    challenge_id: Mapped[int] = mapped_column(ForeignKey("challenges.id"))
    team_name: Mapped[str] = mapped_column(String(255))
    members_count: Mapped[int] = mapped_column(Integer)
    skills: Mapped[str] = mapped_column(Text)
    github_url: Mapped[str | None] = mapped_column(Text, nullable=True)
    motivation: Mapped[str] = mapped_column(Text)
    status: Mapped[str] = mapped_column(String(30), default="PENDING")
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.utcnow)

    challenge: Mapped["Challenge"] = relationship(back_populates="applications")

FIELD_WEIGHTS = {
    "problem": 15,
    "goal": 15,
    "target_users": 10,
    "available_data": 15,
    "expected_result": 15,
    "success_metrics": 15,
    "constraints": 5,
    "deadline": 10,
}

def has_meaningful_value(value) -> bool:
    if value is None:
        return False
    if isinstance(value, str):
        return len(value.strip()) >= 3
    return bool(value)

def calculate_readiness(challenge: dict) -> dict:
    score = 0
    completed = []
    missing = []

    for field, weight in FIELD_WEIGHTS.items():
        if has_meaningful_value(challenge.get(field)):
            score += weight
            completed.append(field)
        else:
            missing.append(field)

    return {
        "score": min(score, 100),
        "completed_fields": completed,
        "missing_fields": missing,
        "weights": FIELD_WEIGHTS,
    }

FastAPI application.

Required endpoints:

GET  /api/health
POST /api/challenges/analyze
POST /api/challenges
GET  /api/challenges
GET  /api/challenges/{id}
POST /api/challenges/{id}/clarify
POST /api/challenges/{id}/technical-review
POST /api/challenges/{id}/publish
POST /api/challenges/{id}/applications
GET  /api/challenges/{id}/applications
PATCH /api/applications/{id}/select

Core rules:

1. /analyze
   - accepts raw description
   - calls OpenAI
   - returns structured analysis + deterministic readiness

2. POST /challenges
   - persists analyzed challenge
   - status DRAFT/CLARIFYING/READY

3. /clarify
   - accepts answers by field
   - updates only allowed challenge fields
   - recalculates readiness

4. /technical-review
   - calls NVIDIA if configured
   - failure must be graceful

5. /publish
   - MVP threshold: readiness >= 70
   - changes status to PUBLISHED

6. /applications
   - only published challenges accept applications

7. /select
   - selected application -> SELECTED
   - other applications for same challenge -> REJECTED
   - persist result

http://localhost:5173

from app.services.readiness import calculate_readiness

def test_empty_challenge_is_zero():
    result = calculate_readiness({})
    assert result["score"] == 0

def test_complete_challenge_is_100():
    challenge = {
        "problem": "Manual feedback analysis is too slow.",
        "goal": "Automate feedback analysis.",
        "target_users": "University administration.",
        "available_data": "10000 anonymized student reviews.",
        "expected_result": "Dashboard and API.",
        "success_metrics": "F1 score >= 0.85.",
        "constraints": "No personal data.",
        "deadline": "2026-11-30",
    }
    result = calculate_readiness(challenge)
    assert result["score"] == 100
    assert result["missing_fields"] == []

def test_partial_challenge():
    challenge = {
        "problem": "Manual analysis is slow.",
        "goal": "Automate analysis.",
    }
    result = calculate_readiness(challenge)
    assert result["score"] == 30
    assert "available_data" in result["missing_fields"]
